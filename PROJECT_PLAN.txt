      * Compare password (plaintext comparison for now)
      * Generate JWT token with payload: { user_id, email, role }

   * Error: 401 if token invalid/expired
3. POST /api/auth/logout (optional for now)
   * Just return success (client will delete token)

CREATE TABLE trips (
    id SERIAL PRIMARY KEY,
    client_name VARCHAR(255) NOT NULL,
    pickup_location VARCHAR(255) NOT NULL,
    dropoff_location VARCHAR(255) NOT NULL,
    pickup_lat DECIMAL(10, 8),
    pickup_lng DECIMAL(11, 8),
    dropoff_lat DECIMAL(10, 8),
    dropoff_lng DECIMAL(11, 8),
    scheduled_start TIMESTAMP NOT NULL,
    scheduled_end TIMESTAMP,
    actual_start TIMESTAMP,
    actual_end TIMESTAMP,
    status VARCHAR(50) DEFAULT 'scheduled', -- 'scheduled', 'in_progress', 'completed', 'cancelled'
    flight_number VARCHAR(50),
    airline VARCHAR(100),
    assigned_agent_id INTEGER REFERENCES users(id),
    assigned_parent_id INTEGER REFERENCES users(id),
    assigned_clinician_id INTEGER REFERENCES users(id),
    created_by_id INTEGER REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


-- Location tracking table
CREATE TABLE location_updates (
    id SERIAL PRIMARY KEY,
    trip_id INTEGER REFERENCES trips(id) ON DELETE CASCADE,
    agent_id INTEGER REFERENCES users(id),
    latitude DECIMAL(10, 8) NOT NULL,
    longitude DECIMAL(11, 8) NOT NULL,
    accuracy DECIMAL(10, 2),
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    source VARCHAR(50) DEFAULT 'gps' -- 'gps' or 'flight'
);


-- Documents table
CREATE TABLE documents (
    id SERIAL PRIMARY KEY,
    trip_id INTEGER REFERENCES trips(id) ON DELETE CASCADE,
    uploaded_by_id INTEGER REFERENCES users(id),
    filename VARCHAR(255) NOT NULL,
    file_url VARCHAR(500) NOT NULL,
    file_type VARCHAR(50),
    file_size INTEGER,
    uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


-- Messages table
CREATE TABLE messages (
    id SERIAL PRIMARY KEY,
    trip_id INTEGER REFERENCES trips(id) ON DELETE CASCADE,
    sender_id INTEGER REFERENCES users(id),
    message_text TEXT NOT NULL,
    sent_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    read_by_recipient BOOLEAN DEFAULT FALSE
);


-- Client notes table (for dashboard)
CREATE TABLE client_notes (
    id SERIAL PRIMARY KEY,
    trip_id INTEGER REFERENCES trips(id) ON DELETE CASCADE,
    created_by_id INTEGER REFERENCES users(id),
    note_text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


-- Indexes for performance

CREATE INDEX idx_trips_status ON trips(status);
CREATE INDEX idx_location_trip ON location_updates(trip_id, timestamp DESC);
CREATE INDEX idx_documents_trip ON documents(trip_id);
CREATE INDEX idx_messages_trip ON messages(trip_id, sent_at);

// MapComponent.jsx
import React, { useEffect, useRef } from 'react';


const TripMap = ({ pickup, dropoff }) => {
  const mapRef = useRef(null);
  
  useEffect(() => {
    // Initialize map
    const map = new google.maps.Map(mapRef.current, {
      zoom: 10,
      center: pickup
    });
    
    // Add pickup marker (green)
    const pickupMarker = new google.maps.Marker({
      position: pickup,
      map: map,
      label: 'P',
      title: 'Pickup Location',
      icon: 'http://maps.google.com/mapfiles/ms/icons/green-dot.png'
    });
    
    // Add dropoff marker (red)
    const dropoffMarker = new google.maps.Marker({
      position: dropoff,
      map: map,
      label: 'D',
      title: 'Dropoff Location',
      icon: 'http://maps.google.com/mapfiles/ms/icons/red-dot.png'
    });
    
    // Draw route
    const directionsService = new google.maps.DirectionsService();
    const directionsRenderer = new google.maps.DirectionsRenderer({
      map: map,
      suppressMarkers: true // We have custom markers
    });
    
    directionsService.route({
      origin: pickup,
      destination: dropoff,
      travelMode: 'DRIVING'
    }, (result, status) => {
      if (status === 'OK') {
        directionsRenderer.setDirections(result);
      }
    });
    
    // Auto-fit bounds to show both markers
    const bounds = new google.maps.LatLngBounds();
    bounds.extend(pickup);
    bounds.extend(dropoff);
    map.fitBounds(bounds);
    
  }, [pickup, dropoff]);
  
  return <div ref={mapRef} style={{ width: '100%', height: '400px' }} />;
};


export default TripMap;

    (position) => {

    {

      sendLocationToBackend(latitude, longitude, accuracy);

  return { watchId, intervalId };

Process:
python
def post_location(trip_id, request):
    # 1. Verify user is authenticated
    user = get_current_user(request)
    
    # 2. Get trip from database
    trip = db.trips.get(trip_id)
    if not trip:
        return 404
    
    # 3. Verify user is the assigned agent
    if trip.assigned_agent_id != user.id and user.role != 'admin':
        return 403  # Only assigned agent can update location
    
    # 4. Validate input
    if not (-90 <= request.latitude <= 90):
        return 400, "Invalid latitude"
    if not (-180 <= request.longitude <= 180):
        return 400, "Invalid longitude"
    
    # 5. Insert location update
    location = db.location_updates.create({
        'trip_id': trip_id,
        'agent_id': user.id,
        'latitude': request.latitude,
        'longitude': request.longitude,
        'accuracy': request.accuracy,
        'timestamp': request.timestamp or datetime.now(),
        'source': 'gps'
    })
    
    # 6. Update trip status if not already in progress
    if trip.status == 'scheduled':
        db.trips.update(trip_id, {
            'status': 'in_progress',
            'actual_start': datetime.now()
        })
    
    return 200, {'success': True, 'location_id': location.id}

import React, { useEffect, useRef, useState } from 'react';


const LiveTripMap = ({ tripId, pickup, dropoff, isLive }) => {
  const mapRef = useRef(null);
  const [currentLocation, setCurrentLocation] = useState(null);
  const [locationHistory, setLocationHistory] = useState([]);
  const [lastUpdate, setLastUpdate] = useState(null);
  const agentMarkerRef = useRef(null);
  const pathLineRef = useRef(null);
  
  // Fetch latest location
  const fetchLocation = async () => {
    try {
      const response = await fetch(`/api/trips/${tripId}/location/latest`, {
        headers: { 'Authorization': `Bearer ${token}` }
      });
      const data = await response.json();
      
      if (data.latitude && data.longitude) {
        setCurrentLocation({
          lat: data.latitude,
          lng: data.longitude
        });
        setLastUpdate(new Date(data.timestamp));
        
        // Add to history for breadcrumb trail
        setLocationHistory(prev => [...prev, {
          lat: data.latitude,
          lng: data.longitude
        }]);
      }
    } catch (error) {
      console.error('Failed to fetch location:', error);
    }
  };
  
  // Poll every 10 seconds
  useEffect(() => {
    if (!isLive) return;
    
    // Fetch immediately
    fetchLocation();
    
    // Then poll every 10 seconds
    const intervalId = setInterval(fetchLocation, 10000);
    
    return () => clearInterval(intervalId);
  }, [tripId, isLive]);
  
  // Initialize map
  useEffect(() => {
    const map = new google.maps.Map(mapRef.current, {
      zoom: 12,
      center: pickup
    });
    
    // Pickup marker (green)
    new google.maps.Marker({
      position: pickup,
      map: map,
      label: 'P',
      icon: 'http://maps.google.com/mapfiles/ms/icons/green-dot.png'
    });
    
    // Dropoff marker (red)
    new google.maps.Marker({
      position: dropoff,
      map: map,
      label: 'D',
      icon: 'http://maps.google.com/mapfiles/ms/icons/red-dot.png'
    });
    
    // Create agent marker (car icon)
    agentMarkerRef.current = new google.maps.Marker({
      map: map,
      icon: {
        url: 'https://maps.google.com/mapfiles/kml/shapes/cabs.png',
        scaledSize: new google.maps.Size(40, 40)
      },
      title: 'Transport Agent'
    });
    
    // Create path line for breadcrumbs
    pathLineRef.current = new google.maps.Polyline({
      map: map,
      strokeColor: '#4285F4',
      strokeOpacity: 0.7,
      strokeWeight: 3
    });
    
    // Store map reference
    mapRef.current = map;
    
  }, [pickup, dropoff]);
  
  // Update agent marker position when location changes
  useEffect(() => {
    if (!currentLocation || !agentMarkerRef.current) return;
    
    // Animate marker to new position
    agentMarkerRef.current.setPosition(currentLocation);
    
    // Update breadcrumb trail
    if (pathLineRef.current) {
      pathLineRef.current.setPath(locationHistory);
    }
    
    // Center map on agent
    if (mapRef.current) {
      mapRef.current.panTo(currentLocation);
    }
    
  }, [currentLocation, locationHistory]);
  
  return (
    <div>
      <div ref={mapRef} style={{ width: '100%', height: '400px' }} />
      
      {lastUpdate && (
        <div style={{ marginTop: '10px', color: '#666' }}>
          Last updated: {getTimeAgo(lastUpdate)}
        </div>
      )}
      
      {isLive && (
        <div style={{ marginTop: '5px', color: '#4CAF50' }}>
          🟢 Live tracking active
        </div>
      )}
    </div>
  );
};


// Helper function
function getTimeAgo(date) {
  const seconds = Math.floor((new Date() - date) / 1000);
  
  if (seconds < 60) return `${seconds} seconds ago`;
  const minutes = Math.floor(seconds / 60);
  if (minutes < 60) return `${minutes} minutes ago`;
  const hours = Math.floor(minutes / 60);
  return `${hours} hours ago`;
}

# Install: pip install python-socketio


from socketio import AsyncServer, ASGIApp
from fastapi import FastAPI


app = FastAPI()
sio = AsyncServer(async_mode='asgi', cors_allowed_origins='*')
socket_app = ASGIApp(sio, app)


# Store connected clients by trip_id
connected_clients = {}  # { trip_id: [sid1, sid2, ...] }


@sio.event
async def connect(sid, environ):
    print(f'Client connected: {sid}')


@sio.event
async def disconnect(sid):
    print(f'Client disconnected: {sid}')
    # Remove from all rooms
    for trip_id, clients in connected_clients.items():
        if sid in clients:
            clients.remove(sid)


@sio.event
async def subscribe_trip(sid, data):
    """Client subscribes to updates for a specific trip"""
    trip_id = data['trip_id']
    token = data['token']
    
    # Verify token and access to trip
    user = verify_jwt(token)
    trip = db.trips.get(trip_id)
    
    if not has_trip_access(user, trip):
        await sio.emit('error', {'message': 'Access denied'}, to=sid)
        return
    
    # Add to room for this trip
    await sio.enter_room(sid, f'trip_{trip_id}')
    
    if trip_id not in connected_clients:
        connected_clients[trip_id] = []
    connected_clients[trip_id].append(sid)
    
    print(f'Client {sid} subscribed to trip {trip_id}')
    
    # Send current location immediately
    latest_location = db.location_updates.get_latest(trip_id)
    if latest_location:
        await sio.emit('location_update', {
            'latitude': latest_location.latitude,
            'longitude': latest_location.longitude,
            'timestamp': latest_location.timestamp.isoformat()
        }, to=sid)


@sio.event
async def unsubscribe_trip(sid, data):
    """Client unsubscribes from trip updates"""
    trip_id = data['trip_id']
    await sio.leave_room(sid, f'trip_{trip_id}')
    
    if trip_id in connected_clients and sid in connected_clients[trip_id]:
        connected_clients[trip_id].remove(sid)

        'longitude': location_data['longitude'],
        'accuracy': location_data['accuracy'],
        'timestamp': location_data['timestamp']

import requests
from datetime import datetime, timedelta


AVIATION_STACK_API_KEY = os.getenv('AVIATION_STACK_API_KEY')


def fetch_flight_status(flight_number):
    """
    Fetch current status of a flight
    Returns: dict with flight info or None if not found
    """
    url = 'http://api.aviationstack.com/v1/flights'
    params = {
        'access_key': AVIATION_STACK_API_KEY,
        'flight_iata': flight_number  # e.g., 'UA1234'
    }
    
    try:
        response = requests.get(url, params=params)
        data = response.json()
        
        if data.get('data') and len(data['data']) > 0:
            flight = data['data'][0]
            
            return {
                'flight_number': flight['flight']['iata'],
                'airline': flight['airline']['name'],
                'status': flight['flight_status'],  # 'scheduled', 'active', 'landed'
                'departure_airport': flight['departure']['airport'],
                'departure_gate': flight['departure']['gate'],
                'departure_time': flight['departure']['actual'] or flight['departure']['scheduled'],
                'arrival_airport': flight['arrival']['airport'],
                'arrival_gate': flight['arrival']['gate'],
                'arrival_time': flight['arrival']['actual'] or flight['arrival']['estimated'],
                'aircraft_lat': flight.get('live', {}).get('latitude'),
                'aircraft_lng': flight.get('live', {}).get('longitude'),
                'aircraft_altitude': flight.get('live', {}).get('altitude'),
                'aircraft_speed': flight.get('live', {}).get('speed_horizontal')
            }
    except Exception as e:
        print(f'Failed to fetch flight status: {e}')
        return None


# Background task to update flight info
async def update_flight_tracking(trip_id):
    """
    Periodically update flight status for active trips
    Run this every 5 minutes
    """
    trip = db.trips.get(trip_id)
    
    if not trip.flight_number:
        return
    
    # Only track if trip is in progress and GPS is stale
    if trip.status != 'in_progress':
        return
    
    latest_gps = db.location_updates.get_latest(trip_id, source='gps')
    gps_age = (datetime.now() - latest_gps.timestamp).seconds if latest_gps else 999999
    
    # If GPS is older than 5 minutes, switch to flight tracking
    if gps_age > 300:
        flight_info = fetch_flight_status(trip.flight_number)
        
        if flight_info and flight_info['aircraft_lat']:
            # Save flight position as location update
            db.location_updates.create({
                'trip_id': trip_id,
                'agent_id': trip.assigned_agent_id,
                'latitude': flight_info['aircraft_lat'],
                'longitude': flight_info['aircraft_lng'],
                'accuracy': 1000,  # Flight tracking less accurate
                'timestamp': datetime.now(),
                'source': 'flight'
            })
            
            # Broadcast to WebSocket clients
            await broadcast_location_update(trip_id, {
                'latitude': flight_info['aircraft_lat'],
                'longitude': flight_info['aircraft_lng'],
                'source': 'flight',
                'flight_info': flight_info
            })

TASK 13: Auto-Switch GPS ↔ Flight Logic
Context: Automatically detect when to switch between GPS tracking (driving) and flight tracking (on plane) without manual intervention.
Requirements:

* Switch to flight tracking if flight number exists

Detection Logic:
python
def determine_tracking_mode(trip_id):
    """
    Decide whether to use GPS or flight tracking
    Returns: 'gps', 'flight', or 'unknown'
    """
    trip = db.trips.get(trip_id)
    
    # Get latest GPS location
    latest_gps = db.location_updates.get_latest(
        trip_id=trip_id,
        source='gps'
    )
    
    # Calculate how long since last GPS update
    if latest_gps:
        gps_age_seconds = (datetime.now() - latest_gps.timestamp).seconds
    else:
        gps_age_seconds = 999999
    
    # Decision tree:
    
    # 1. If GPS is recent (< 5 minutes), use GPS
    if gps_age_seconds < 300:
        return 'gps'
    
    # 2. If GPS is stale AND flight number exists, check flight status
    if trip.flight_number and gps_age_seconds >= 300:
        flight_info = fetch_flight_status(trip.flight_number)
        
        if flight_info:
            # Check if flight is currently active
            if flight_info['status'] in ['active', 'en-route']:
                return 'flight'
            
            # If flight landed recently, wait for GPS to resume
            if flight_info['status'] == 'landed':
                landing_time = parse(flight_info['arrival_time'])
                minutes_since_landing = (datetime.now() - landing_time).seconds / 60
                
                # Give agent 30 minutes to get off plane and resume GPS
                if minutes_since_landing < 30:
                    return 'flight'  # Still show flight tracking
                else:
                    return 'unknown'  # GPS should have resumed by now
    
    # 3. GPS is stale but no flight info
    return 'unknown'

def get_messages(trip_id, request):
    # 1. Verify access
    user = get_current_user(request)
    trip = db.trips.get(trip_id)
    if not has_trip_access(user, trip):
        return 403
    
    # 2. Get query params
    limit = request.query.get('limit', 100)
    since = request.query.get('since')
    
    # 3. Fetch messages
    query = db.messages.filter(trip_id=trip_id)
    if since:
        query = query.filter(sent_at__gt=since)
    
    messages = query.order_by('sent_at').limit(limit).all()
    
    # 4. Join with users table to get sender names
    result = []
    for msg in messages:
        sender = db.users.get(msg.sender_id)
        result.append({
            'id': msg.id,
            'sender': {
                'id': sender.id,
                'name': f'{sender.first_name} {sender.last_name}',
                'role': sender.role
            },
            'message': msg.message_text,
            'sent_at': msg.sent_at.isoformat(),
            'read': msg.read_by_recipient,
            'is_mine': msg.sender_id == user.id
        })
    
    return 200, {
        'messages': result,
        'count': len(result)
    }

import React, { useState, useEffect, useRef } from 'react';


const TripChat = ({ tripId, currentUserId }) => {
  const [messages, setMessages] = useState([]);
  const [newMessage, setNewMessage] = useState('');
  const [loading, setLoading] = useState(false);
  const messagesEndRef = useRef(null);
  
  // Fetch messages on mount
  useEffect(() => {
    fetchMessages();
  }, [tripId]);
  
  const fetchMessages = async () => {
    try {
      const response = await fetch(`/api/trips/${tripId}/messages`, {
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('auth_token')}`
        }
      });
      const data = await response.json();
      setMessages(data.messages);
      scrollToBottom();
    } catch (error) {
      console.error('Failed to fetch messages:', error);
    }
  };
  
  const sendMessage = async (e) => {
    e.preventDefault();
    
    if (!newMessage.trim()) return;
    
    setLoading(true);
    try {
      const response = await fetch(`/api/trips/${tripId}/messages`, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${localStorage.getItem('auth_token')}`,
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({ message: newMessage })
      });
      
      const sentMessage = await response.json();
      
      // Add to messages list
      setMessages(prev => [...prev, sentMessage]);
      
      // Clear input
      setNewMessage('');
      
      // Scroll to bottom
      scrollToBottom();
    } catch (error) {
      console.error('Failed to send message:', error);
      alert('Failed to send message');
    } finally {
      setLoading(false);
    }
  };
  
  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };
  
  const formatTime = (timestamp) => {
    const date = new Date(timestamp);
    const now = new Date();
    const diff = now - date;
    
    // If today, show time only
    if (diff < 86400000) {
      return date.toLocaleTimeString('en-US', { 
        hour: 'numeric', 
        minute: '2-digit' 
      });
    }
    // If within week, show day and time
    else if (diff < 604800000) {
      return date.toLocaleString('en-US', { 
        weekday: 'short', 
        hour: 'numeric', 
        minute: '2-digit' 
      });
    }
    // Otherwise show full date
    else {
      return date.toLocaleString('en-US', { 
        month: 'short', 
        day: 'numeric',
        hour: 'numeric', 
        minute: '2-digit' 
      });
    }
  };
  
  return (
    <div className="trip-chat">
      <div className="chat-header">
        <h3>💬 Trip Messages</h3>
      </div>
      
      <div className="chat-messages">
        {messages.length === 0 && (
          <div className="no-messages">
            No messages yet. Start a conversation!
          </div>
        )}
        
        {messages.map((msg) => (
          <div 
            key={msg.id} 
            className={`message ${msg.is_mine ? 'message-mine' : 'message-other'}`}
          >
            <div className="message-header">
              <span className="sender-name">
                {msg.sender.name}
              </span>
              <span className="sender-role">
                ({msg.sender.role})
              </span>
              <span className="message-time">
                {formatTime(msg.sent_at)}
              </span>
            </div>
            <div className="message-bubble">
              {msg.message}
            </div>
          </div>
        ))}
        
        <div ref={messagesEndRef} />
      </div>
      
      <form className="chat-input" onSubmit={sendMessage}>
        <textarea
          value={newMessage}
          onChange={(e) => setNewMessage(e.target.value)}
          placeholder="Type a message..."
          rows="2"
          disabled={loading}
          onKeyPress={(e) => {
            // Send on Enter (without Shift)
            if (e.key === 'Enter' && !e.shiftKey) {
              e.preventDefault();
              sendMessage(e);
            }
          }}
        />
        <button type="submit" disabled={loading || !newMessage.trim()}>
          {loading ? 'Sending...' : 'Send'}
        </button>
      </form>
    </div>
  );
};


export default TripChat;

async def user_typing(sid, data):

    User started typing - broadcast to others
    """

    token = data['token']

    # Store typing state
    if trip_id not in typing_users:
        typing_users[trip_id] = {}
    
    typing_users[trip_id][user.id] = {
        'name': f'{user.first_name} {user.last_name}',
        'timestamp': datetime.now()
    }
    
    # Broadcast typing status to room (excluding sender)
    await broadcast_typing_status(trip_id, exclude_sid=sid)

def update_trip_status(trip_id, request):
    user = get_current_user(request)
    trip = db.trips.get(trip_id)
    
    # Verify permission (must be assigned agent or admin)
    if user.role != 'admin' and trip.assigned_agent_id != user.id:
        return 403, "Only assigned agent or admin can update status"
    
    # Validate status
    valid_statuses = ['scheduled', 'in_progress', 'completed', 'cancelled']
    new_status = request.status
    if new_status not in valid_statuses:
        return 400, f"Invalid status. Must be one of: {valid_statuses}"
    
    # Get current status
    old_status = trip.status
    
    # Record status change in history
    db.trip_status_history.create({
        'trip_id': trip_id,
        'changed_by_id': user.id,
        'old_status': old_status,
        'new_status': new_status,
        'changed_at': datetime.now(),
        'notes': request.get('notes', '')
    })
    
    # Update trip status
    update_data = {'status': new_status}
    
    # Set timestamps based on status
    if new_status == 'in_progress' and not trip.actual_start:
        update_data['actual_start'] = datetime.now()
    elif new_status == 'completed' and not trip.actual_end:
        update_data['actual_end'] = datetime.now()
    
    db.trips.update(trip_id, update_data)
    
    # Broadcast to WebSocket clients
    await sio.emit('status_changed', {
        'trip_id': trip_id,
        'old_status': old_status,
        'new_status': new_status,
        'changed_by': f'{user.first_name} {user.last_name}',
        'timestamp': datetime.now().isoformat()
    }, room=f'trip_{trip_id}')
    
    return 200, {
        'success': True,
        'trip_id': trip_id,
        'status': new_status
    }

import secrets
from datetime import datetime, timedelta


def generate_tokens(user):
    # Access token (short-lived: 1 hour)
    access_token = jwt.encode({
        'user_id': user.id,
        'email': user.email,
        'role': user.role,
        'exp': datetime.utcnow() + timedelta(hours=1)
    }, JWT_SECRET)
    
    # Refresh token (long-lived: 30 days)
    refresh_token = secrets.token_urlsafe(64)
    
    db.refresh_tokens.create({
        'user_id': user.id,
        'token': refresh_token,
        'expires_at': datetime.utcnow() + timedelta(days=30)
    })
    
    return {
        'access_token': access_token,
        'refresh_token': refresh_token,
        'expires_in': 3600  # 1 hour in seconds
    }


# New endpoint: POST /api/auth/refresh
def refresh_access_token(refresh_token):
    # Verify refresh token
    token_record = db.refresh_tokens.get(token=refresh_token)
    
    if not token_record or token_record.revoked:
        return 401, "Invalid refresh token"
    
    if token_record.expires_at < datetime.utcnow():
        return 401, "Refresh token expired"
    
    # Generate new access token
    user = db.users.get(token_record.user_id)
    access_token = jwt.encode({
        'user_id': user.id,
        'email': user.email,
        'role': user.role,
        'exp': datetime.utcnow() + timedelta(hours=1)
    }, JWT_SECRET)
    
    return 200, {
        'access_token': access_token,
        'expires_in': 3600
    }


# Logout: Revoke refresh token
def logout(refresh_token):
    db.refresh_tokens.update(
        {'token': refresh_token},
        {'revoked': True}
    )

# POST /api/auth/reset-password
def reset_password(token, new_password):
    # Verify token
    token_record = db.password_reset_tokens.get(token=token)
    
    if not token_record or token_record.used:
        return 400, "Invalid or expired reset token"
    
    if token_record.expires_at < datetime.utcnow():
        return 400, "Reset token expired"
    
    # Validate new password
    if len(new_password) < 8:
        return 400, "Password must be at least 8 characters"
    
    # Hash new password
    hashed = bcrypt.hashpw(new_password.encode('utf-8'), bcrypt.gensalt())
    
    # Update user password
    db.users.update(token_record.user_id, {
        'password': hashed.decode('utf-8')
    })
    
    # Mark token as used
    db.password_reset_tokens.update(token_record.id, {'used': True})
    
    # Revoke all refresh tokens for security
    db.refresh_tokens.update(
        {'user_id': token_record.user_id},
        {'revoked': True}
    )
    
    return 200, "Password reset successfully"
